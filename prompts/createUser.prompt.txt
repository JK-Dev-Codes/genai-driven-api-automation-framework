# POST — Create an entity
# And I create a <entity> with name "<Name_>" appended with timestamp and description "<description>"
# Endpoint: https://your-api-base-url/<resource>
# Payload: {
#   "name": "<Name_>",
#   "description": "<description>",
#   "draft": false
# }

# GET — Retrieve a list
# And I GET all <entities>.
# Endpoint: https://your-api-base-url/<resource>?includeDisabled=false

# Chained POST — Create a child entity (references parent ID)
# And I create a <child-entity> under the <parent-entity>.
# Endpoint: https://your-api-base-url/<child-resource>
# Payload: {
#   "name": "<ChildName_>",
#   "description": "<description>",
#   "<parentId>": "<parent-uuid>",
#   "draft": false
# }

# GET — Search
# And I search for the created <entity>.
# Endpoint: https://your-api-base-url/global/search?term={term}&includeDisabled=false

# PUT — Update (with specific field values)
# And I update the <entity>.
# Endpoint: https://your-api-base-url/<resource>/{id}
# | field | value         |
# | name  | <UpdatedName> |

# PUT — Disable
# Disable the <entity>.
# Endpoint: https://your-api-base-url/<resource>/hierarchy/{id}/enabled/false/draft/false
# Inline payload: { "enabled": false, "draft": false }
